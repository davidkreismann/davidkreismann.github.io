\newcommand{\resumeItem}[1]{\item\small{#1}}

\newcommand{\resumeSubheading}[4]{
  \vspace{-2pt}\item
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} & #2 \\
      \textit{\small#3} & \textit{\small #4} \\
    \end{tabular*}\vspace{-7pt}
}

\newcommand{\resumeSubSubheading}[4]{
  \item
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textit{\small #1} & \textit{\small #2} \\
      \textit{\footnotesize #3} & \textit{\footnotesize #4} \\
    \end{tabular*}\vspace{-7pt}
}

\newcommand{\resumeWideHeading}[4]{
  \vspace{-2pt}\item
    \begin{tabular*}{0.97\textwidth}[t]{p{0.80\textwidth}@{\extracolsep{\fill}}r}
      {\small\textbf{#1}} & {\small #2} \\
      {\small\textit{#3}} & {\small\textit{#4}} \\
    \end{tabular*}\vspace{-7pt}
}

\renewcommand\labelitemii{$\vcenter{\hbox{\tiny$\bullet$}}$}

\newcommand{\resumeSmallHeading}[4]{
  \vspace{-2pt}\item
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
      {\small\textbf{#1}} & {\small #2} \\
      {\small\textit{#3}} & {\small\textit{#4}} \\
    \end{tabular*}\vspace{-7pt}
}

\section{Education}
\resumeSubHeadingListStart
{%- for e in site.data.cv.education %}
  \resumeSubheading{ {%- include latex.html text=e.org -%} }{ {%- include latex.html text=e.location -%} }{ {%- include latex.html text=e.title -%} }{ {%- include latex.html text=e.date -%} }
  {%- assign items = e.pdf_details | default: e.details %}
  {%- if items %}
  \resumeItemListStart
    {%- for d in items %}
    \resumeItem{ {%- include latex.html text=d -%} }
    {%- endfor %}
  \resumeItemListEnd
  {%- endif %}
{%- endfor %}
\resumeSubHeadingListEnd

\section{Experience}
\resumeSubHeadingListStart
{%- for e in site.data.cv.experience %}
  \resumeSubheading{ {%- include latex.html text=e.title -%} }{ {%- include latex.html text=e.date -%} }{ {%- include latex.html text=e.org -%} }{ {%- include latex.html text=e.location -%} }
  {%- if e.details %}
  \resumeItemListStart
    {%- for d in e.details %}
    \resumeItem{ {%- include latex.html text=d -%} }
    {%- endfor %}
  \resumeItemListEnd
  {%- endif %}
  {%- for r in e.roles %}
    {%- assign parts = r.title | split: ", " %}
    \resumeSubSubheading{ {%- include latex.html text=parts.first -%} }{ {%- include latex.html text=r.date -%} }{ {%- assign rest = parts | shift | join: ", " -%}{%- include latex.html text=rest -%} }{ {%- include latex.html text=r.location -%} }
    \resumeItemListStart
      \resumeItem{ {%- include latex.html text=r.text -%} }
    \resumeItemListEnd
  {%- endfor %}
{%- endfor %}
\resumeSubHeadingListEnd
